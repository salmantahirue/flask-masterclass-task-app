What is an ORM?

ORM (Object-Relational Mapping) lets you work with the database using Python classes (models) instead of raw SQL. You define models (e.g. Task), relationships (e.g. user has many tasks), and use db.session to add, query, and commit.

# Model
class Task(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    title = db.Column(db.String(200), nullable=False)
    user_id = db.Column(db.Integer, db.ForeignKey("user.id"))
    user = db.relationship("User", backref=db.backref("tasks", lazy="dynamic"))

# Query
tasks = Task.query.filter_by(user_id=current_user.id).order_by(Task.updated_at.desc()).all()
db.session.add(task)
db.session.commit()
